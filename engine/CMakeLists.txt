cmake_minimum_required(VERSION 3.25)

project(agri-engine
  VERSION 0.1.0
  DESCRIPTION "AgriRobots execution engine: the C++ core the robots run"
  LANGUAGES CXX)

# C++23, no compiler extensions: the robot image and the CI image have to build
# the same language, and a GNU extension that one toolchain lacks is a build that
# only works on somebody's laptop.
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
  set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type" FORCE)
endif()

option(AGRI_ENGINE_BUILD_TESTS "Build the equivalence test suite" ON)
option(AGRI_ENGINE_WERROR "Treat compiler warnings as errors" OFF)
option(AGRI_ENGINE_INSTALL "Add install rules so the ROS 2 executor can link this library" ON)

add_library(agri_engine STATIC
  src/canonical_json.cpp
  src/sha256.cpp
  src/trace.cpp)
add_library(agri::engine ALIAS agri_engine)

target_include_directories(agri_engine
  PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
         $<INSTALL_INTERFACE:include>)

# Vendored single-header dependencies. SYSTEM, so their warnings are not this
# repository's to fix and -Werror does not turn them into build failures.
target_include_directories(agri_engine SYSTEM
  PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/third_party>)

target_compile_features(agri_engine PUBLIC cxx_std_23)

if(MSVC)
  target_compile_options(agri_engine PRIVATE /W4 /permissive-)
  if(AGRI_ENGINE_WERROR)
    target_compile_options(agri_engine PRIVATE /WX)
  endif()
else()
  target_compile_options(agri_engine PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
  if(AGRI_ENGINE_WERROR)
    target_compile_options(agri_engine PRIVATE -Werror)
  endif()
endif()

if(AGRI_ENGINE_BUILD_TESTS)
  enable_testing()

  # Catch2 in its own target: the amalgamated source is vendored, so it is built
  # with default warnings and never with -Werror.
  add_library(catch2_amalgamated STATIC third_party/catch2/catch_amalgamated.cpp)
  target_include_directories(catch2_amalgamated SYSTEM
    PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/third_party/catch2)

  # No test_main.cpp: the vendored Catch2 amalgamation already provides main().
  # Defining CATCH_AMALGAMATED_CUSTOM_MAIN on the catch2 target would hand it
  # back, which is what to do if the suite ever needs custom start-up.
  add_executable(agri_engine_tests
    test/test_canonical_json.cpp
    test/test_golden_traces.cpp
    test/test_sha256.cpp)
  target_link_libraries(agri_engine_tests PRIVATE agri_engine catch2_amalgamated)
  target_include_directories(agri_engine_tests PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/test)

  # The fixtures are generated by the TypeScript reference engine and committed;
  # tests read them from the source tree, never from the build tree, so a build
  # directory cannot end up testing fixtures of its own.
  target_compile_definitions(agri_engine_tests
    PRIVATE AGRI_TESTDATA_DIR="${CMAKE_CURRENT_SOURCE_DIR}/testdata")

  if(NOT MSVC)
    target_compile_options(agri_engine_tests PRIVATE -Wall -Wextra -Wpedantic -Wshadow)
    if(AGRI_ENGINE_WERROR)
      target_compile_options(agri_engine_tests PRIVATE -Werror)
    endif()
  endif()

  add_test(NAME agri_engine_unit COMMAND agri_engine_tests "~[golden]")
  add_test(NAME agri_engine_golden_traces COMMAND agri_engine_tests "[golden]")
endif()

if(AGRI_ENGINE_INSTALL)
  include(GNUInstallDirs)
  install(TARGETS agri_engine EXPORT agri-engine-targets
    ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
  install(DIRECTORY include/agri DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
  install(EXPORT agri-engine-targets
    NAMESPACE agri::
    DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/agri-engine)
endif()
